134 [[nodiscard]] static bool activationFunctionsEqual(
135 const std::unique_ptr<ActivationFunction>& lhs,
136 const std::unique_ptr<ActivationFunction>& rhs)
137 {
138 if (!lhs || !rhs) {
139 return !lhs && !rhs;
140 }
141 if (lhs->type != rhs->type) {
142 return false;
143 }
144 switch (lhs->type)
145 {
146 case ActivationFunctionType::SIGMOID:
147 {
148 const auto* l = dynamic_cast<const SigmoidFunction*>(lhs.get());
149 const auto* r = dynamic_cast<const SigmoidFunction*>(rhs.get());
150 return l != nullptr && r != nullptr && (*l == *r);
151 }
152 case ActivationFunctionType::HEAVISIDE:
153 {
154 const auto* l = dynamic_cast<const HeavisideFunction*>(lhs.get());
155 const auto* r = dynamic_cast<const HeavisideFunction*>(rhs.get());
156 return l != nullptr && r != nullptr && (*l == *r);
157 }
158 case ActivationFunctionType::ABSSIGMOID:
159 {
160 const auto* l = dynamic_cast<const AbsSigmoidFunction*>(lhs.get());
161 const auto* r = dynamic_cast<const AbsSigmoidFunction*>(rhs.get());
162 return l != nullptr && r != nullptr && (*l == *r);
163 }
164 default:
165 return false;
166 }
